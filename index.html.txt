    <script>
    (() => {
        'use strict';

        const STORAGE_KEY = 'SEQWHITE_STUDIO_ACCOUNTING_DATA_V2';

        // 服務與分類選單項目 (不帶價格)
        const INCOME_SERVICES = [
            '純色', '跳色', '漸層', '貓眼', '法式', '鏡面', 
            '客製款', '延甲', '卸甲', '穿戴甲', '夜間服務+500', '補甲', '其他收入'
        ];

        const EXPENSE_CATEGORIES = [
            '材料耗材', '房租', '水電', '廣告行銷', '設備工具', '運費', '手續費', '其他支出'
        ];

        // 深度 DOM 選擇器簡化
        const $ = id => document.getElementById(id);

        // 工具函式
        const getTodayStr = () => {
            const d = new Date();
            return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
        };

        const formatMoney = num => 'NT$ ' + Number(num || 0).toLocaleString('zh-TW');
        const getMonthStr = dateStr => dateStr.slice(0, 7);
        const escapeHtml = str => String(str ?? '').replace(/[&<>"']/g, c => ({
            '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
        }[c]));

        // 應用程式狀態
        let records = [];
        let selectedDate = getTodayStr();
        let viewMonth = getMonthStr(selectedDate);
        let activePage = 'calendar';

        // 載入資料
        function loadRecords() {
            try {
                const saved = localStorage.getItem(STORAGE_KEY);
                if (saved) {
                    const parsed = JSON.parse(saved);
                    if (Array.isArray(parsed)) records = parsed.filter(isValidRecord);
                }
            } catch (e) {
                alert('無法讀取本機紀錄，請確認瀏覽器隱私設定。');
            }
        }

        function isValidRecord(r) {
            return r && typeof r.id === 'string' &&
                /^\d{4}-\d{2}-\d{2}$/.test(r.date) &&
                ['income', 'expense'].includes(r.type) &&
                Array.isArray(r.items) &&
                Number.isFinite(Number(r.amount)) && Number(r.amount) >= 0;
        }

        function saveRecords() {
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(records));
                return true;
            } catch (e) {
                alert('儲存失敗：本機儲存空間不足。');
                return false;
            }
        }

        // 初始化選單 DOM
        function initFormOptions() {
            // 生成收入 Chips (複選)
            const chipsContainer = $('incomeChipsContainer');
            chipsContainer.innerHTML = INCOME_SERVICES.map((service, idx) => `
                <div class="chip-checkbox">
                    <input type="checkbox" id="chip_${idx}" value="${escapeHtml(service)}">
                    <label for="chip_${idx}" class="chip-label">${escapeHtml(service)}</label>
                </div>
            `).join('');

            // 生成支出 Select (單選)
            const expenseSelect = $('expenseCategorySelect');
            expenseSelect.innerHTML = EXPENSE_CATEGORIES.map(cat => `
                <option value="${escapeHtml(cat)}">${escapeHtml(cat)}</option>
            `).join('');
        }

        // 計算統計數值
        function calculateSummary(dataList) {
            return dataList.reduce((acc, r) => {
                const amt = Number(r.amount) || 0;
                if (r.type === 'income') acc.income += amt;
                else acc.expense += amt;
                return acc;
            }, { income: 0, expense: 0 });
        }

        function renderSummaryHtml(summary) {
            const net = summary.income - summary.expense;
            return `
                <div class="summary-grid">
                    <div class="metric-card">
                        <small>總收入</small>
                        <strong class="text-income">${formatMoney(summary.income)}</strong>
                    </div>
                    <div class="metric-card">
                        <small>總支出</small>
                        <strong class="text-expense">${formatMoney(summary.expense)}</strong>
                    </div>
                    <div class="metric-card">
                        <small>淨利</small>
                        <strong>${formatMoney(net)}</strong>
                    </div>
                </div>
            `;
        }

        // 渲染條目 HTML
        function renderEntryItemHtml(r) {
            const isIncome = r.type === 'income';
            const itemsTags = (r.items || []).map(i => `<span class="tag">${escapeHtml(i)}</span>`).join('');
            
            return `
                <div class="entry-item">
                    <div class="entry-info">
                        <div class="entry-tags">${itemsTags}</div>
                        <div class="entry-sub">${escapeHtml(r.date)} · ${escapeHtml(r.payment || '現金')}${r.note ? ' · ' + escapeHtml(r.note) : ''}</div>
                    </div>
                    <div class="entry-amount ${isIncome ? 'text-income' : 'text-expense'}">
                        ${isIncome ? '+' : '−'}${formatMoney(r.amount)}
                    </div>
                    <div class="entry-actions">
                        <button class="action-btn" onclick="openModalForEdit('${r.id}')" title="編輯">✎</button>
                        <button class="action-btn" onclick="deleteRecord('${r.id}')" title="刪除">🗑️</button>
                    </div>
                </div>
            `;
        }

        // 頁面 1：月曆渲染
        function renderCalendar() {
            const [y, m] = viewMonth.split('-').map(Number);
            const firstDayIndex = new Date(y, m - 1, 1).getDay();
            const totalDays = new Date(y, m, 0).getDate();

            $('monthTitle').textContent = `${y} 年 ${m} 月`;

            let daysHtml = Array(firstDayIndex).fill('<span></span>').join('');

            for (let d = 1; d <= totalDays; d++) {
                const dateStr = `${viewMonth}-${String(d).padStart(2, '0')}`;
                const hasRecord = records.some(r => r.date === dateStr);
                const isSelected = dateStr === selectedDate;
                const isToday = dateStr === getTodayStr();

                daysHtml += `
                    <button class="day-cell ${isSelected ? 'selected' : ''} ${isToday ? 'today' : ''}" 
                            data-date="${dateStr}">
                        ${d}
                        ${hasRecord ? '<i class="dot"></i>' : ''}
                    </button>
                `;
            }

            $('calendarDays').innerHTML = daysHtml;
            $('selectedDateTitle').textContent = selectedDate.replaceAll('-', ' / ') + ' 收支明細';

            // 渲染選取日期的條目
            const dayRecords = records
                .filter(r => r.date === selectedDate)
                .sort((a, b) => (b.created || '').localeCompare(a.created || ''));

            $('dayEntries').innerHTML = dayRecords.length
                ? dayRecords.map(renderEntryItemHtml).join('')
                : '<div class="empty-state">當日尚無收支紀錄</div>';

            $('daySummary').innerHTML = renderSummaryHtml(calculateSummary(dayRecords));
        }

        // 頁面 2：總紀錄渲染
        function renderRecordsPage() {
            const monthVal = $('filterMonth').value;
            const typeVal = $('filterType').value;
            const searchVal = $('filterSearch').value.trim().toLowerCase();

            let filtered = records.filter(r => {
                if (monthVal && getMonthStr(r.date) !== monthVal) return false;
                if (typeVal !== 'all' && r.type !== typeVal) return false;
                if (searchVal) {
                    const matchItem = (r.items || []).some(i => i.toLowerCase().includes(searchVal));
                    const matchNote = (r.note || '').toLowerCase().includes(searchVal);
                    const matchPayment = (r.payment || '').toLowerCase().includes(searchVal);
                    if (!matchItem && !matchNote && !matchPayment) return false;
                }
                return true;
            });

            filtered.sort((a, b) => b.date.localeCompare(a.date) || (b.created || '').localeCompare(a.created || ''));

            $('recordsSummary').innerHTML = renderSummaryHtml(calculateSummary(filtered));
            $('allEntriesList').innerHTML = filtered.length
                ? filtered.map(renderEntryItemHtml).join('')
                : '<div class="empty-state">無符合條件的紀錄</div>';
        }

        // 頁面 3：報表渲染
        function renderReportsPage() {
            const monthVal = $('reportMonth').value;
            const monthRecords = records.filter(r => getMonthStr(r.date) === monthVal);
            const summary = calculateSummary(monthRecords);
            const maxVal = Math.max(1, summary.income, summary.expense);

            $('reportSummary').innerHTML = renderSummaryHtml(summary);

            // 收支比例圖
            $('reportBarRatio').innerHTML = `
                <div class="bar-item">
                    <span>總收入</span>
                    <div class="bar-track"><div class="bar-fill" style="width: ${(summary.income / maxVal) * 100}%"></div></div>
                    <span style="text-align: right">${formatMoney(summary.income)}</span>
                </div>
                <div class="bar-item">
                    <span>總支出</span>
                    <div class="bar-track"><div class="bar-fill expense-fill" style="width: ${(summary.expense / maxVal) * 100}%"></div></div>
                    <span style="text-align: right">${formatMoney(summary.expense)}</span>
                </div>
            `;

            // 分類排行榜
            renderCategoryRank(monthRecords, 'income', $('incomeCategoryBars'));
            renderCategoryRank(monthRecords, 'expense', $('expenseCategoryBars'));

            // 近 6 個月趨勢圖
            renderTrendChart(monthVal);
        }

        function renderCategoryRank(dataList, type, container) {
            const filtered = dataList.filter(r => r.type === type);
            const catMap = {};

            filtered.forEach(r => {
                (r.items || []).forEach(item => {
                    catMap[item] = (catMap[item] || 0) + Number(r.amount);
                });
            });

            const sorted = Object.entries(catMap).sort((a, b) => b[1] - a[1]);
            const maxVal = Math.max(1, ...sorted.map(s => s[1]));

            container.innerHTML = sorted.length ? sorted.map(([name, amt]) => `
                <div class="bar-item">
                    <span>${escapeHtml(name)}</span>
                    <div class="bar-track">
                        <div class="bar-fill ${type === 'expense' ? 'expense-fill' : ''}" style="width: ${(amt / maxVal) * 100}%"></div>
                    </div>
                    <span style="text-align: right">${formatMoney(amt)}</span>
                </div>
            `).join('') : '<div class="empty-state">暫無資料</div>';
        }

        function renderTrendChart(currentMonthStr) {
            const [y, m] = currentMonthStr.split('-').map(Number);
            const months = Array.from({ length: 6 }, (_, i) => {
                const d = new Date(y, m - 1 - (5 - i), 1);
                return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
            });

            const monthData = months.map(mStr => {
                const subRecords = records.filter(r => getMonthStr(r.date) === mStr);
                return { month: mStr, ...calculateSummary(subRecords) };
            });

            const maxVal = Math.max(1, ...monthData.flatMap(d => [d.income, d.expense]));

            $('trendBars').innerHTML = monthData.map(d => `
                <div class="bar-item">
                    <span>${d.month.slice(2)}</span>
                    <div>
                        <div class="bar-track" style="margin-bottom: 3px;">
                            <div class="bar-fill" style="width: ${(d.income / maxVal) * 100}%"></div>
                        </div>
                        <div class="bar-track">
                            <div class="bar-fill expense-fill" style="width: ${(d.expense / maxVal) * 100}%"></div>
                        </div>
                    </div>
                    <span style="text-align: right">${formatMoney(d.income - d.expense)}</span>
                </div>
            `).join('') + '<p style="font-size:10px; color:var(--text-muted); margin-top:8px; text-align:right;">深色：收入｜淺色：支出｜右側：淨利</p>';
        }

        // 全域 Render
        function renderApp() {
            renderCalendar();
            renderRecordsPage();
            renderReportsPage();
        }

        // 切換頁面
        function switchPage(pageKey) {
            activePage = pageKey;
            document.querySelectorAll('.toolbar-btn').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.page === pageKey);
            });

            ['calendar', 'records', 'reports'].forEach(pk => {
                $(`page-${pk}`).classList.toggle('hidden', pk !== pageKey);
            });

            renderApp();
        }

        // Modal 操作邏輯
        function openModal(editId = null) {
            $('entryForm').reset();
            $('editEntryId').value = editId || '';

            // 清空 Checkbox 狀態
            document.querySelectorAll('#incomeChipsContainer input').forEach(cb => cb.checked = false);

            if (editId) {
                const r = records.find(x => x.id === editId);
                if (r) {
                    $('formModalTitle').textContent = '編輯收支紀錄';
                    $('entryType').value = r.type;
                    $('entryDate').value = r.date;
                    $('entryAmount').value = r.amount;
                    $('entryPayment').value = r.payment || '現金';
                    $('entryNote').value = r.note || '';

                    toggleEntryTypeUI(r.type);

                    if (r.type === 'income') {
                        (r.items || []).forEach(item => {
                            const cb = Array.from(document.querySelectorAll('#incomeChipsContainer input'))
                                            .find(el => el.value === item);
                            if (cb) cb.checked = true;
                        });
                    } else {
                        if (r.items && r.items[0]) $('expenseCategorySelect').value = r.items[0];
                    }
                }
            } else {
                $('formModalTitle').textContent = '新增收支紀錄';
                $('entryType').value = 'income';
                $('entryDate').value = selectedDate;
                toggleEntryTypeUI('income');
            }

            $('entryModal').classList.add('active');
        }

        function closeModal() {
            $('entryModal').classList.remove('active');
        }

        function toggleEntryTypeUI(type) {
            if (type === 'income') {
                $('incomeServicesGroup').classList.remove('hidden');
                $('expenseCategoryGroup').classList.add('hidden');
            } else {
                $('incomeServicesGroup').classList.add('hidden');
                $('expenseCategoryGroup').classList.remove('hidden');
            }
        }

        // 全域暴露編輯與刪除給動態 HTML
        window.openModalForEdit = id => openModal(id);

        window.deleteRecord = id => {
            const r = records.find(x => x.id === id);
            if (r && confirm(`確定要刪除「${(r.items || []).join(' + ')}」${formatMoney(r.amount)} 的紀錄嗎？`)) {
                records = records.filter(x => x.id !== id);
                saveRecords();
                renderApp();
            }
        };

        // 事件綁定
        document.querySelectorAll('.toolbar-btn').forEach(btn => {
            btn.onclick = () => switchPage(btn.dataset.page);
        });

        $('calendarDays').onclick = e => {
            const btn = e.target.closest('[data-date]');
            if (btn) {
                selectedDate = btn.dataset.date;
                renderCalendar();
            }
        };

        $('btnPrevMonth').onclick = () => shiftMonth(-1);
        $('btnNextMonth').onclick = () => shiftMonth(1);

        function shiftMonth(delta) {
            const [y, m] = viewMonth.split('-').map(Number);
            const d = new Date(y, m - 1 + delta, 1);
            viewMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
            selectedDate = `${viewMonth}-01`;
            renderCalendar();
        }

        $('btnAddCalendar').onclick = () => openModal();
        $('btnAddRecords').onclick = () => openModal();
        $('btnCloseModal').onclick = closeModal;
        $('btnCancelModal').onclick = closeModal;

        $('entryType').onchange = e => toggleEntryTypeUI(e.target.value);

        // 表單提交
        $('entryForm').onsubmit = e => {
            e.preventDefault();

            const type = $('entryType').value;
            const amount = Number($('amount' in e.target ? e.target.amount.value : $('entryAmount').value));

            if (!Number.isSafeInteger(amount) || amount < 0) {
                alert('請輸入有效的整數金額');
                return;
            }

            let selectedItems = [];
            if (type === 'income') {
                selectedItems = Array.from(document.querySelectorAll('#incomeChipsContainer input:checked'))
                                     .map(cb => cb.value);
                if (selectedItems.length === 0) {
                    alert('請至少勾選一項美甲服務項目');
                    return;
                }
            } else {
                selectedItems = [$('expenseCategorySelect').value];
            }

            const editId = $('editEntryId').value;
            const recordObj = {
                id: editId || (crypto.randomUUID ? crypto.randomUUID() : Date.now() + '-' + Math.random()),
                date: $('entryDate').value,
                type,
                items: selectedItems,
                amount,
                payment: $('entryPayment').value,
                note: $('entryNote').value.trim(),
                created: records.find(x => x.id === editId)?.created || new Date().toISOString()
            };

            if (editId) {
                records = records.map(x => x.id === editId ? recordObj : x);
            } else {
                records.push(recordObj);
            }

            if (saveRecords()) {
                selectedDate = recordObj.date;
                viewMonth = getMonthStr(selectedDate);
                closeModal();
                renderApp();
            }
        };

        // 篩選監聽
        ['filterMonth', 'filterType', 'filterSearch'].forEach(id => {
            $(id).addEventListener('input', renderRecordsPage);
        });

        $('reportMonth').addEventListener('input', renderReportsPage);

        // JSON 備份與匯入
        $('btnExportJSON').onclick = () => {
            const dataStr = JSON.stringify({
                app: 'SEQWHITE.STUDIO',
                version: 2,
                exportedAt: new Date().toISOString(),
                records
            }, null, 2);

            const blob = new Blob([dataStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `SEQWHITE_Backup_${getTodayStr()}.json`;
            a.click();
            setTimeout(() => URL.revokeObjectURL(url), 1000);
        };

        $('btnImportJSON').onclick = () => $('importFileInput').click();

        $('importFileInput').onchange = async e => {
            const file = e.target.files[0];
            if (!file) return;

            try {
                const content = await file.text();
                const json = JSON.parse(content);

                if (json.app !== 'SEQWHITE.STUDIO' || !Array.isArray(json.records) || !json.records.every(isValidRecord)) {
                    throw new Error('檔案格式不相容或已損壞');
                }

                if (confirm(`確定要匯入 ${json.records.length} 筆紀錄並與現有資料合併嗎？`)) {
                    const recordMap = new Map(records.map(r => [r.id, r]));
                    json.records.forEach(r => recordMap.set(r.id, r));
                    records = Array.from(recordMap.values());

                    saveRecords();
                    renderApp();
                    alert('匯入備份成功！');
                }
            } catch (err) {
                alert('無法匯入檔案：' + err.message);
            } finally {
                e.target.value = '';
            }
        };

        // 初始化預設值
        $('filterMonth').value = viewMonth;
        $('reportMonth').value = viewMonth;
        initFormOptions();
        loadRecords();
        renderApp();
    })();
    </script>
